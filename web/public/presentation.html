# AI Agent: autonomous card purchase
result = subprocess.run(
  ["***REMOVED***", "card", "buy",
   "--amount", "50",
   "--format", "json"],
  capture_output=True
)

data = json.loads(result.stdout)
if data["ok"]:
    card = data["data"]
    print(f"Card ready: {card['last4']}")
else:
    error = data["error"]
    print(f"Fix: {error['suggestion']}")
